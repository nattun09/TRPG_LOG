■概要
log_export.htmlに、位置選択できるヘッダーを追加する。
飛べる個所は、画像、任意のテキストが対象。
(最新版)
・level-1、level-2の見出しを追加している場合、そこに飛ぶシーンを追加する
・BGMの追加も可能


■実装方法

①bodyタグの上に以下を追加する（最新版対応。level1、level-2の見出しならすべて対応可能。level1は太字になる。）
※BGMの追加をしたい場合は最終行の箇所にファイル名追加！　例：<audio id="bgm" src="***.mp3" loop></audio>


<script>
document.addEventListener("DOMContentLoaded", () => {
  /* ----- セレクトボックス処理 ----- */
  const select = document.getElementById("image-selector");
  const elements = document.querySelectorAll(
    ".heading-item.export.level-1.log-item, .heading-item.export.level-2.log-item"
  );
  elements.forEach((el, index) => {
    let id = el.id;
    if (!id) {
      const level = el.classList.contains("level-1") ? "l1" : "l2";
      id = `scene-${level}-${index}`;
      el.id = id;
    }
    const label = el.textContent.trim();
    const option = document.createElement("option");
    option.value = id;
    option.textContent = label;
    if (el.classList.contains("level-1")) option.style.fontWeight = "bold";
    select.appendChild(option);
  });
  select.addEventListener("change", function () {
    const target = document.getElementById(this.value);
    if (target) target.scrollIntoView({ behavior: "smooth", block: "center" });
  });

  /* ----- 音楽ボタン処理 ----- */
const audioEl = document.getElementById("bgm");
const btn = document.getElementById("music-toggle");
const volumeBar = document.getElementById("volume-bar");

// 音声ファイルが空の場合はボタンとバーを非表示
const audioSrc = audioEl.getAttribute("src"); // ←これなら空文字を正確に取得可能
if (!audioSrc || audioSrc.trim() === "") {
  btn.style.display = "none";
  volumeBar.style.display = "none";
} else {
  btn.addEventListener("click", async () => {
    try {
      if (audioEl.paused) {
        await audioEl.play();
        btn.innerHTML = "❚❚";
        btn.classList.add("active");
      } else {
        audioEl.pause();
        btn.innerHTML = "▶";
        btn.classList.remove("active");
      }
    } catch (err) {
      console.log("再生エラー:", err);
    }
  });

  volumeBar.addEventListener("input", () => {
    audioEl.volume = parseFloat(volumeBar.value);
  });
}
});
</script>
